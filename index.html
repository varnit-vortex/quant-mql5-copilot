FastAPI & MetaTrader 5 VPS Bridge Script

Deploy this microservice to connect LangGraph agent webhooks directly to a VPS MetaTrader 5 terminal.

# vortex_mt5_bridge.py
import MetaTrader5 as mt5
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI(title="VortexQuant MT5 Bridge")

class OrderSignal(BaseModel):
    symbol: str = "XAUUSD"
    action: str  # "BUY" or "SELL"
    volume: float = 0.1
    sl_points: int = 250
    tp_points: int = 500
    magic: int = 882109

@app.on_event("startup")
def init_mt5():
    if not mt5.initialize():
        raise RuntimeError("MT5 Init failed")
    print("✓ MetaTrader 5 Connected.")

@app.post("/execute_trade")
def execute(sig: OrderSignal):
    point = mt5.symbol_info(sig.symbol).point
    price = mt5.symbol_info_tick(sig.symbol).ask if sig.action == "BUY" else mt5.symbol_info_tick(sig.symbol).bid
    req = {
        "action": mt5.TRADE_ACTION_DEAL,
        "symbol": sig.symbol,
        "volume": sig.volume,
        "type": mt5.ORDER_TYPE_BUY if sig.action == "BUY" else mt5.ORDER_TYPE_SELL,
        "price": price,
        "sl": price - (sig.sl_points * point) if sig.action == "BUY" else price + (sig.sl_points * point),
        "tp": price + (sig.tp_points * point) if sig.action == "BUY" else price - (sig.tp_points * point),
        "magic": sig.magic,
        "comment": "VortexQuant_SMC_Auto",
    }
    res = mt5.order_send(req)
    return {"status": "SUCCESS", "order": res.order}
