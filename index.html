Python MetaTrader 5 Bridge Script

FastAPI + MT5

Deploy this lightweight Python microservice to execute orders directly from LangGraph or webhook signals to a VPS-hosted MetaTrader 5 terminal. 

# quant_mt5_bridge.py
import MetaTrader5 as mt5
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI(title="QuantAI MT5 Bridge")

class OrderSignal(BaseModel):
    symbol: str = "XAUUSD"
    action: str  # "BUY" or "SELL"
    volume: float = 0.1
    sl_points: int = 250
    tp_points: int = 500
    magic: int = 882109

@app.on_event("startup")
def init_mt5():
    if not mt5.initialize():
        raise RuntimeError(f"MT5 Init failed: {mt5.last_error()}")
    print("✓ MetaTrader 5 Terminal connected successfully.")

@app.post("/execute_trade")
def execute_trade(sig: OrderSignal):
    symbol_info = mt5.symbol_info(sig.symbol)
    if not symbol_info.visible:
        mt5.symbol_select(sig.symbol, True)
        
    point = symbol_info.point
    price = mt5.symbol_info_tick(sig.symbol).ask if sig.action == "BUY" else mt5.symbol_info_tick(sig.symbol).bid
    sl = price - (sig.sl_points * point) if sig.action == "BUY" else price + (sig.sl_points * point)
    tp = price + (sig.tp_points * point) if sig.action == "BUY" else price - (sig.tp_points * point)
    
    request = {
        "action": mt5.TRADE_ACTION_DEAL,
        "symbol": sig.symbol,
        "volume": sig.volume,
        "type": mt5.ORDER_TYPE_BUY if sig.action == "BUY" else mt5.ORDER_TYPE_SELL,
        "price": price,
        "sl": sl,
        "tp": tp,
        "magic": sig.magic,
        "comment": "QuantAI_Autonomous_Execution",
        "type_time": mt5.ORDER_TIME_GTC,
        "type_filling": mt5.ORDER_FILLING_IOC,
    }
    
    result = mt5.order_send(request)
    if result.retcode != mt5.TRADE_RETCODE_DONE:
        raise HTTPException(status_code=400, detail=f"Order failed: {result.comment}")
    return {"status": "SUCCESS", "order_ticket": result.order, "price": result.price}
